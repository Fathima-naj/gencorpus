JavaScript Built-in Objects

1. String Object

The String object is used to work with text. Strings are sequences of characters like letters, numbers, symbols, and spaces.

Creating Strings

let name = "Fathima";
let city = 'Malappuram';
let message = `Hello World`;

length

Returns number of characters in a string.

let text = "JavaScript";

console.log(text.length);
Output 
10

charAt()

Returns character at a specific position.

let text = "JavaScript";

console.log(text.charAt(2));
Output 
v

toUpperCase()

let text = "hello";

console.log(text.toUpperCase());
Output 
HELLO

toLowerCase()

let text = "HELLO";

console.log(text.toLowerCase());
Output 
hello

includes()

Checks whether a string contains a value.

let text = "JavaScript";

console.log(text.includes("Script"));
Output 
true

indexOf()

Returns the position of first occurrence.

let text = "Hello World";

console.log(text.indexOf("World"));
Output 
6

startsWith()

let text = "JavaScript";

console.log(text.startsWith("Java"));
Output 
true

endsWith()

let text = "JavaScript";

console.log(text.endsWith("Script"));
Output 
true

slice()

Extracts part of a string.

let text = "JavaScript";

console.log(text.slice(0,4));
Output 
Java

substring()

let text = "JavaScript";

console.log(text.substring(4,10));
Output 
Script

replace()

Replaces first occurrence.

let text = "Hello World";

console.log(text.replace("World","JavaScript"));
Output 
Hello JavaScript

replaceAll()

let text = "Hello Hello";

console.log(text.replaceAll("Hello","Hi"));
Output 
Hi Hi

trim()

Removes spaces from both sides.

let text = "   hello   ";

console.log(text.trim());
Output 
hello

split()

Converts string to array.

let fruits = "apple,banana,orange";

console.log(fruits.split(","));
Output 
["apple","banana","orange"]

concat()

Joins strings.

let first = "Hello";
let second = "World";

console.log(first.concat(" ",second));
Output 
Hello World

repeat()

let text = "Hi ";

console.log(text.repeat(3));
Output 
Hi Hi Hi

padStart()

let num = "5";

console.log(num.padStart(3,"0"));
Output 
005

padEnd()

let num = "5";

console.log(num.padEnd(3,"0"));
Output 
500

4. Math Object

The Math object provides built-in mathematical constants and functions. Unlike other objects, you do not need to create it using new.

Math.round()

Rounds a number to the nearest integer.

console.log(Math.round(4.6));
Output 
5

Math.floor()

Rounds a number down to the nearest integer.

console.log(Math.floor(4.9));
Output 
4

Math.ceil()

Rounds a number up to the nearest integer.

console.log(Math.ceil(4.1));
Output 
5

Math.trunc()

Removes the decimal part of a number.

console.log(Math.trunc(4.9));
Output 
4

Math.max()

Returns the largest value.

console.log(Math.max(5,10,2,8));
Output 
10

Math.min()

Returns the smallest value.

console.log(Math.min(5,10,2,8));
Output 
2

Math.abs()

Returns the absolute (positive) value.

console.log(Math.abs(-10));
Output 
10

Math.sqrt()

Returns the square root of a number.

console.log(Math.sqrt(25));
Output 
5

Math.pow()

Returns power of a number.

console.log(Math.pow(2,3));
Output 
8

Math.random()

Generates a random number between 0 and 1.

console.log(Math.random());
Example: random number between 1 and 10 
let random = Math.floor(Math.random() * 10) + 1;

console.log(random);

Math.PI

Returns the value of π (pi).

console.log(Math.PI);
Output 
3.141592653589793
